get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
               # * 1000 kWh to MWH 
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

load_ntd_energy <- function(){
  library(readxl)
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    library(httr2)
  
    request("https://www.transit.dot.gov/") |>
      req_url_path("sites",
                   "fta.dot.gov",
                   "files",
                   "2026-09",
                   "2025 Energy Consumption_260806.xlsx") |> 
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_ENERGY_FILE)
  }
  
  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
        cat("I was unable to download the NTD Energy File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_energy.xlsx')` and try again.")
    }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |> 
    select(-c(`Reporter Type`, 
              `Reporting Module`, 
              `Gallons of Other Fuel`, 
              `Other Fuel Description`)) |>
    mutate(across(-c(`Agency Name`, 
                     `Mode`,
                     `TOS`), 
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum), 
              .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

NTD_ENERGY <- load_ntd_energy()

Task 1c: NTD Service Provided

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    library(httr2)
  
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query`=paste(
        "SELECT  `_5_digit_ntd_id` AS `NTD_ID`", 
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000", 
        sep=",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_SERVICE_FILE)
  }
  
  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
        cat("I was unable to download the NTD Service File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.xlsx')` and try again.")
    }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

NTD_SERVICE <- load_ntd_service()

Task 5: Calculate Emissions

# Step 1: NTD_SERVICE can have several rows per Agency + Mode (one per type of service),
# so we total them to get one row per Agency + Mode, matching NTD_ENERGY
SERVICE_BY_MODE <- NTD_SERVICE |>
  mutate(NTD_ID = as.numeric(NTD_ID)) |> # NTD_ENERGY stores this ID as a number
  group_by(NTD_ID, state, Mode) |>
  summarize(UPT = sum(total_upt, na.rm = TRUE),
            PMT = sum(sum_passenger_miles, na.rm = TRUE),
            .groups = "drop")

# Step 2: join energy use + service (by agency ID and mode), then + state CO2 rate.
# inner_join: we only keep rows where we know both fuel use and ridership,
# and only states in the EIA data (so DC and territories are dropped)
EMISSIONS <- NTD_ENERGY |>
  select(-`Bio-Diesel Description`) |>
  inner_join(SERVICE_BY_MODE,
             join_by(`NTD ID` == NTD_ID, Mode == Mode)) |>
  inner_join(EIA_SEP_REPORT |> select(abbreviation, CO2_MWh),
             join_by(state == abbreviation))

# Step 3: pounds of CO2 per unit of fuel, hard-coded from the EIA
# (https://www.eia.gov/environment/emissions/co2_vol_mass.php), aligned to NTD fuel names by best effort
CO2_DIESEL    <- 22.45  # Diesel and Home Heating Fuel (Distillate Fuel Oil), per gallon
CO2_BIODIESEL <- 22.45  # Not listed by EIA; treated as diesel (most transit bio-diesel is a diesel blend)
CO2_BUNKER    <- 24.78  # Residual Fuel Oil, per gallon
CO2_GASOLINE  <- 18.73  # Finished Motor Gasoline, per gallon
CO2_ETHANOL   <- 18.73  # Not listed by EIA; treated as finished motor gasoline (ethanol blend)
CO2_KEROSENE  <- 21.78  # Kerosene, per gallon
CO2_LPG       <- 12.68  # Propane, per gallon
CO2_CNG       <- 120.85 * 126.67 / 1000  # Natural Gas is 120.85 per 1000 cubic ft; 1 gallon equivalent of CNG = 126.67 cubic ft
CO2_LNG       <- 120.85 * 82.6 / 1000    # 1 gallon of LNG = about 82.6 cubic ft of natural gas
CO2_HYDROGEN  <- 0      # Hydrogen produces no CO2 when used

# Step 4: total emissions (in pounds of CO2) for each Agency + Mode
EMISSIONS <- EMISSIONS |>
  mutate(total_emissions_lbs =
           `Gallons of Diesel Fuel` * CO2_DIESEL +
           `Gallons of Bio-Diesel` * CO2_BIODIESEL +
           `Gallons of Bunker Fuel` * CO2_BUNKER +
           `Gallons of Gasoline` * CO2_GASOLINE +
           `Gallons of Ethanol` * CO2_ETHANOL +
           `Gallons of Kerosene` * CO2_KEROSENE +
           `Gallons of Liquified Petroleum Gas` * CO2_LPG +
           `Gallons of Compressed Natural Gas` * CO2_CNG +
           `Gallons of Liquified Nat Gas` * CO2_LNG +
           `Kilograms of Hydrogen` * CO2_HYDROGEN +
           # electricity: kWh / 1000 = MWh, times the state's pounds of CO2 per MWh
           (`Kilowatt hours of Electric Battery` + `Kilowatt hours of Electric Propulsion`) / 1000 * CO2_MWh)

EMISSIONS |>
  select(`Agency Name`, Mode, state, UPT, PMT, total_emissions_lbs) |>
  slice_max(total_emissions_lbs, n = 10) |>
  gt() |>
  cols_label(`Agency Name` = "Agency",
             state = "State",
             UPT = "Unlinked Passenger Trips",
             PMT = "Passenger Miles",
             total_emissions_lbs = "Total CO2 Emissions (lbs)") |>
  fmt_number(columns = c(UPT, PMT, total_emissions_lbs), decimals = 0) |>
  tab_header(title = md("**Highest-Emitting Transit Services**"),
             subtitle = "Agency + Mode pairs, 2025") |>
  tab_source_note("Sources: National Transit Database (2025 Energy Consumption, 2025 Service by Agency); U.S. EIA State Electricity Profiles and CO2 Emissions Coefficients")

How Do Agencies Improve These Numbers?

The data shows us that some city transportation agencies are doing a much better job at lowering their CO2 emissions per passenger mile than others. The easiest way for other agencies to mimick this is to get more people riding public transport (trains, buses, shuttles), since the vehicle burns the same amount of fuel regardless. This way CO2 per passenger mile is decreased. Agencies should focus on improving the overall quality of service, which is likely to result in higher usage.
